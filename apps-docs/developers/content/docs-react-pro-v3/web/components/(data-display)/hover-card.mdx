import {HoverCard} from "@vezham/react-pro-v3";

function ImageCard() {
  return (
    <>
      <img
        alt="Neo Brutalism design"
        className="h-36 w-full rounded-t-2xl object-cover"
        src="https://heroui-assets.nyc3.cdn.digitaloceanspaces.com/heroui-pro-landing/chat-message.webp"
      />

        <p className="text-sm font-semibold">Neo Brutalism</p>
        <p className="text-muted mt-1 text-sm leading-relaxed">
          Raw aesthetics, bold colors, and playful typography.
        </p>
      </div>
    </>
  );
}

const delays = [
  {closeDelay: 0, label: "Instant", openDelay: 0},
  {closeDelay: 300, label: "Default (700 / 300)", openDelay: 700},
  {closeDelay: 500, label: "Slow (1s / 500)", openDelay: 1000},
];

export function CustomDelays() {
  return (
    <div className="flex items-center gap-6 px-10 py-24">
      {delays.map(({closeDelay, label, openDelay}) => (
        <HoverCard key={label} closeDelay={closeDelay} openDelay={openDelay}>
          <HoverCard.Trigger>
            <Button variant="outline">{label}</Button>
          </HoverCard.Trigger>
          <HoverCard.Content className="w-72 p-0">
            <HoverCard.Arrow />
            <ImageCard />
          </HoverCard.Content>
        </HoverCard>
      ))}
    </div>
  );
}

## CSS Classes

### Element Classes

* `.hover-card__trigger` - Inline wrapper around the trigger element
* `.hover-card__content` - The floating panel that appears on hover
* `.hover-card__arrow` - Optional pointing arrow connecting content to trigger

### Interactive States

* **Entering/Exiting**: `[data-entering="true"]` / `[data-exiting="true"]` on `.hover-card__content` (fade + zoom + slide animation). Skipped when swapping between cards during the shared warm-up window.
* **Placement**: `[data-placement="top|bottom|left|right"]` on `.hover-card__content` and `.hover-card__arrow` (directional animations and arrow rotation)

## Accessibility

HoverCard is built on RAC [PreviewTrigger](https://react-spectrum.adobe.com/react-aria/PreviewTrigger.html), so it works across input methods:

* **Pointer**: hovering the trigger opens the card after `openDelay`. A safe area between the trigger and the card keeps it open while the pointer travels toward it, even diagonally.
* **Keyboard**: focusing the trigger opens the card, `Tab` moves focus into it, and `Escape` closes it.
* **Touch**: long pressing the trigger opens the card and moves the screen reader cursor into it.

The card renders with `role="dialog"`. Pass `aria-label` to `HoverCard.Content` to give it an accessible name.

Delays are coordinated globally with other hover cards and tooltips: once one has opened, the next one opens immediately instead of waiting out its `openDelay` again.

## API Reference

### HoverCard

The root component. Wraps RAC [PreviewTrigger](https://react-spectrum.adobe.com/react-aria/PreviewTrigger.html) to manage hover state and open/close timing.

| Prop           | Type                      | Default | Description                                         |
| -------------- | ------------------------- | ------- | --------------------------------------------------- |
| `open`         | `boolean`                 | -       | Controlled open state                               |
| `defaultOpen`  | `boolean`                 | -       | Default open state (uncontrolled)                   |
| `onOpenChange` | `(open: boolean) => void` | -       | Callback when open state changes                    |
| `openDelay`    | `number`                  | `700`   | Time in ms before opening after hover               |
| `closeDelay`   | `number`                  | `300`   | Time in ms before closing after pointer/focus leave |
| `isDisabled`   | `boolean`                 | `false` | Prevents the hover card from opening                |
| `children`     | `ReactNode`               | -       | Trigger and Content elements                        |

### HoverCard.Trigger

Inline wrapper that tracks hover, focus, and long press on its children.

Also supports all native `span` HTML attributes.

### HoverCard.Content

The floating panel. Wraps RAC [Popover](https://react-spectrum.adobe.com/react-aria/Popover.html).

| Prop        | Type        | Default | Description                               |
| ----------- | ----------- | ------- | ----------------------------------------- |
| `placement` | `Placement` | `'top'` | Popover placement relative to the trigger |
| `offset`    | `number`    | `8`     | Distance from the trigger                 |

Also supports all RAC [Popover](https://react-spectrum.adobe.com/react-aria/Popover.html) props except the open state ones (`isOpen`, `defaultOpen`, `onOpenChange`) and `triggerRef`, which are owned by the root.

### HoverCard.Arrow

Optional pointing arrow. Wraps RAC [OverlayArrow](https://react-spectrum.adobe.com/react-aria/OverlayArrow.html). Renders a default SVG arrow when no children are provided.

| Prop       | Type        | Default | Description          |
| ---------- | ----------- | ------- | -------------------- |
| `children` | `ReactNode` | -       | Custom arrow element |

Also supports all RAC [OverlayArrow](https://react-spectrum.adobe.com/react-aria/OverlayArrow.html) props.
