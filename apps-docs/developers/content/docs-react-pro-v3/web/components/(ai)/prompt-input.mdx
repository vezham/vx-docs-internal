````tsx
"use client";

import {ArrowUp, Paperclip} from "@gravity-ui/icons";
import {useState} from "react";
````

```tsx
import {useEffect, useRef, useState} from "react";

export type PendingAttachment = {
  id: string;
  mimeType?: string;
  name: string;
  src?: string;
};

function createAttachmentId(file: File): string {
  return `${file.name}-${file.lastModified}-${
    globalThis.crypto?.randomUUID?.() ?? Math.random().toString(36).slice(2)
  }`;
}

function revokeAttachmentUrl(attachment: PendingAttachment) {
  if (attachment.src?.startsWith("blob:")) URL.revokeObjectURL(attachment.src);
}

function revokeAttachmentUrls(attachments: readonly PendingAttachment[]) {
  attachments.forEach(revokeAttachmentUrl);
}

export function useDemoAttachments() {
  const [attachments, setAttachments] = useState<PendingAttachment[]>([]);
  const attachmentsRef = useRef<PendingAttachment[]>([]);

  useEffect(() => {
    attachmentsRef.current = attachments;
  }, [attachments]);

  useEffect(() => {
    return () => revokeAttachmentUrls(attachmentsRef.current);
  }, []);

  const clearAttachments = () => {
    setAttachments((current) => {
      revokeAttachmentUrls(current);

      return [];
    });
  };

  const handleFilesSelected = (files: File[]) => {
    setAttachments((current) => [
      ...current,
      ...files.map((file) => ({
        id: createAttachmentId(file),
        mimeType: file.type,
        name: file.name,
        src: file.type.startsWith("image/") ? URL.createObjectURL(file) : undefined,
      })),
    ]);
  };

  const removeAttachment = (id: string) => {
    setAttachments((current) => {
      const removed = current.find((attachment) => attachment.id === id);

      if (removed) revokeAttachmentUrl(removed);

      return current.filter((attachment) => attachment.id !== id);
    });
  };

  const replaceAttachments = (next: PendingAttachment[]) => {
    setAttachments((current) => {
      revokeAttachmentUrls(current);

      return next.map((attachment) => ({...attachment}));
    });
  };

  return {attachments, clearAttachments, handleFilesSelected, removeAttachment, replaceAttachments};
```
```tsx
}

    });
  };

  const handleFilesSelected = (files: File[]) => {
    setAttachments((current) => [
      ...current,
      ...files.map((file) => ({
        id: createAttachmentId(file),
        mimeType: file.type,
        name: file.name,
        src: file.type.startsWith("image/") ? URL.createObjectURL(file) : undefined,
      })),
    ]);
  };

  const removeAttachment = (id: string) => {
    setAttachments((current) => {
      const removed = current.find((attachment) => attachment.id === id);

      if (removed) revokeAttachmentUrl(removed);

      return current.filter((attachment) => attachment.id !== id);
    });
  };

  const replaceAttachments = (next: PendingAttachment[]) => {
    setAttachments((current) => {
      revokeAttachmentUrls(current);

      return next.map((attachment) => ({...attachment}));
    });
  };

  return {attachments, clearAttachments, handleFilesSelected, removeAttachment, replaceAttachments};
```
````tsx
}
````

## Run State

Prompt input supports `submitted`, `streaming`, `ready`, and `error` states for send/stop behavior.

<ComponentPreview name="pro-prompt-input-run-state">

````tsx
"use client";

import type {ChatStatus} from "@vezham/react-pro-v3";

import {useCallback, useEffect, useRef, useState} from "react";

import {PromptInput} from "@vezham/react-pro-v3";

export function RunState() {
  const [value, setValue] = useState("");
  const [status, setStatus] = useState<ChatStatus>("ready");
  const timersRef = useRef<number[]>([]);

  const clearTimers = useCallback(() => {
    timersRef.current.forEach((timer) => window.clearTimeout(timer));
    timersRef.current = [];
  }, []);

  useEffect(() => {
    return () => clearTimers();
  }, [clearTimers]);

  const handleStop = () => {
    clearTimers();
    setStatus("ready");
  };

  const handleSubmit = () => {
    const trimmed = value.trim();

    if (!trimmed || status !== "ready") return;

    setValue("");
    setStatus("submitted");
    clearTimers();

    timersRef.current.push(
      window.setTimeout(() => setStatus("streaming"), 400),
      window.setTimeout(() => setStatus("ready"), 1800),
    );
  };

  return (
    <div className="w-full max-w-[640px]">
      <PromptInput
        status={status}
        value={value}
        onStop={handleStop}
        onSubmit={handleSubmit}
        onValueChange={setValue}
      >
        <PromptInput.Shell>
          <PromptInput.Content>
            <PromptInput.TextArea placeholder="Send to see submitted → streaming → ready" />
          </PromptInput.Content>
          <PromptInput.Toolbar>
            <PromptInput.ToolbarEnd>
              <PromptInput.Send />
            </PromptInput.ToolbarEnd>
          </PromptInput.Toolbar>
        </PromptInput.Shell>
        <PromptInput.Footer>
          Status-driven send button with stop while generating.
        </PromptInput.Footer>
````

```tsx
import {useEffect, useRef, useState} from "react";

export type PendingAttachment = {
  id: string;
  mimeType?: string;
  name: string;
  src?: string;
};

function createAttachmentId(file: File): string {
  return `${file.name}-${file.lastModified}-${
    globalThis.crypto?.randomUUID?.() ?? Math.random().toString(36).slice(2)
  }`;
}

function revokeAttachmentUrl(attachment: PendingAttachment) {
  if (attachment.src?.startsWith("blob:")) URL.revokeObjectURL(attachment.src);
}

function revokeAttachmentUrls(attachments: readonly PendingAttachment[]) {
  attachments.forEach(revokeAttachmentUrl);
}

export function useDemoAttachments() {
  const [attachments, setAttachments] = useState<PendingAttachment[]>([]);
  const attachmentsRef = useRef<PendingAttachment[]>([]);

  useEffect(() => {
    attachmentsRef.current = attachments;
  }, [attachments]);

  useEffect(() => {
    return () => revokeAttachmentUrls(attachmentsRef.current);
  }, []);

  const clearAttachments = () => {
    setAttachments((current) => {
      revokeAttachmentUrls(current);

      return [];
    });
  };

  const handleFilesSelected = (files: File[]) => {
    setAttachments((current) => [
      ...current,
      ...files.map((file) => ({
        id: createAttachmentId(file),
        mimeType: file.type,
        name: file.name,
        src: file.type.startsWith("image/") ? URL.createObjectURL(file) : undefined,
      })),
    ]);
  };

  const removeAttachment = (id: string) => {
    setAttachments((current) => {
      const removed = current.find((attachment) => attachment.id === id);

      if (removed) revokeAttachmentUrl(removed);

      return current.filter((attachment) => attachment.id !== id);
    });
  };

  const replaceAttachments = (next: PendingAttachment[]) => {
    setAttachments((current) => {
      revokeAttachmentUrls(current);

      return next.map((attachment) => ({...attachment}));
    });
  };

  return {attachments, clearAttachments, handleFilesSelected, removeAttachment, replaceAttachments};
}
```

## Token Input

Use `PromptInput.TokenInput` instead of `PromptInput.TextArea` when the prompt should carry inline tokens such as `@mentions` or `/commands`. It is built on the React Aria Components `TokenField`, so its value is a `TokenFieldValue` segment list rather than a string, and it renders a `contenteditable` element instead of a textarea.

Drive it with `usePromptInputTokens`, which tracks the value and resolves the trigger the caret currently sits in. Wrap both the field and `PromptInput.TokenSuggestions` in `PromptInput.Tokens` so arrow keys and Enter move through the menu while the caret stays in the field. The query ends at the first space unless you pass `allowsSpaces`, and the menu closes when the caret moves out of the trigger or the field loses focus. Escape dismisses the menu for the current trigger until the caret leaves it; every other close, including `close()`, only hides the menu until the next edit or caret move.

The suggestion menu is anchored to the caret, not to the composer. For a composer docked to the bottom of the viewport, pass `placement="top start"` so the menu opens upward instead of covering the toolbar.

<Callout type="warning">
  `PromptInput.Tokens` is a React Aria `Autocomplete`, so every menu and list box rendered inside it
  joins the suggestion collection. Keep `PromptInput.Toolbar` and any other menus outside it.
</Callout>

<Callout type="warning">
  `TokenField` is an alpha React Aria Components API. Its behavior and props may change in a future
  release.
</Callout>

The token field owns the prompt and flattens it into a string for `PromptInput.Send` and the compact layout. Leave `value` on `PromptInput` unset when you use it — clear the prompt with `tokens.setValue(new TokenFieldValue([]))` instead.

```tsx
  return (
    <PromptInput onSubmit={() => tokens.setValue(new TokenFieldValue([]))}>
      <PromptInput.Shell>
        <PromptInput.Tokens>
          <PromptInput.Content>
            <PromptInput.TokenInput tokens={tokens} />
          </PromptInput.Content>
          <PromptInput.TokenSuggestions tokens={tokens}>
            <Menu items={items}>
              {(item) => (
                <Menu.Item id={item.id} onAction={() => tokens.insertToken(`@${item.username}`)}>
                  {item.name}
                </Menu.Item>
              )}
            </Menu>
          </PromptInput.TokenSuggestions>
        </PromptInput.Tokens>
        <PromptInput.Toolbar>
          <PromptInput.ToolbarEnd>
            <PromptInput.Send />
          </PromptInput.ToolbarEnd>
        </PromptInput.Toolbar>
      </PromptInput.Shell>
    </PromptInput>
  );
}
```

## API Reference

### PromptInput

| Prop             | Type                                               | Default     | Description                                   |
| ---------------- | -------------------------------------------------- | ----------- | --------------------------------------------- |
| `value`          | `string`                                           | internal    | Controlled input value                        |
| `onValueChange`  | `(value: string) => void`                          | -           | Called when the text changes                  |
| `onSubmit`       | `() => void`                                       | -           | Called when submit is requested               |
| `onStop`         | `() => void`                                       | -           | Called by the send button in stoppable states |
| `status`         | `'ready' \| 'submitted' \| 'streaming' \| 'error'` | `'ready'`   | Composer run state                            |
| `isDisabled`     | `boolean`                                          | `false`     | Disable composer controls                     |
| `lockInputOnRun` | `boolean`                                          | `true`      | Disable textarea while submitted/streaming    |
| `maxHeight`      | `number \| string`                                 | `240`       | Max input height, also as a CSS variable      |
| `size`           | `'sm' \| 'md' \| 'lg'`                             | `'md'`      | Composer size                                 |
| `variant`        | `'primary' \| 'secondary'`                         | `'primary'` | Surface styling variant                       |
| `layout`         | `'stacked' \| 'compact' \| 'inline'`               | `'stacked'` | Composer layout and toolbar placement         |

### PromptInput.TextArea

Extends HeroUI `TextArea` props.

| Prop              | Type      | Default | Description                         |
| ----------------- | --------- | ------- | ----------------------------------- |
| `disableAutosize` | `boolean` | `false` | Disable automatic height adjustment |

### PromptInput.TokenInput

An alternative to `PromptInput.TextArea` that renders inline tokens. Extends React Aria Components `TokenInput` props. Enter submits and Shift+Enter inserts a newline, and Enter is ignored while the composer is running unless `allowSubmitWhileRunning` is set, matching `PromptInput.TextArea`.

| Prop             | Type                                      | Default                       | Description                                            |
| ---------------- | ----------------------------------------- | ----------------------------- | ------------------------------------------------------ |
| `tokens`         | `PromptInputTokenState`                   | -                             | Required token state from `usePromptInputTokens`       |
| `children`       | `(segment: TokenSegment) => ReactElement` | `PromptInput.Token`           | Renders each token segment                             |
| `placeholder`    | `string`                                  | `'What do you want to know?'` | Text shown while the field is empty                    |
| `allowsNewlines` | `boolean`                                 | `true`                        | Whether Shift+Enter inserts a newline                  |
| `isReadOnly`     | `boolean`                                 | `false`                       | Make the field read only                               |
| `isDisabled`     | `boolean`                                 | inherited                     | Defaults to the composer's disabled and run-lock state |

### PromptInput.Tokens

Connects `PromptInput.TokenInput` to `PromptInput.TokenSuggestions` so arrow keys and Enter move through the menu while the caret stays in the field. Wrap both with it, and nothing else — it is a React Aria `Autocomplete`, whose props it extends, so any other menu inside it joins the suggestion collection. Renders no DOM of its own.

### PromptInput.TokenSuggestions

A suggestion menu anchored to the caret of the active trigger. Extends React Aria Components `Popover` props except `triggerRef` and `getTargetRect`, which it sets from the token state.

| Prop        | Type                    | Default          | Description                                      |
| ----------- | ----------------------- | ---------------- | ------------------------------------------------ |
| `tokens`    | `PromptInputTokenState` | -                | Required token state from `usePromptInputTokens` |
| `isOpen`    | `boolean`               | `tokens.isOpen`  | Overrides open state                             |
| `placement` | `Placement`             | `'bottom start'` | Placement relative to the caret                  |

The popover hides itself while its menu has no items, so narrowing `isOpen` is optional.

### PromptInput.Token

A non-editable inline token. Extends React Aria Components `Token` props. Exposes `[data-selected]` and `[data-disabled]` for styling.

### usePromptInputTokens

Tracks a token field value and resolves the trigger the caret currently sits in.

| Option         | Type                               | Default      | Description                                    |
| -------------- | ---------------------------------- | ------------ | ---------------------------------------------- |
| `triggers`     | `string[]`                         | `['@', '/']` | Strings that open the menu at a word boundary  |
| `allowsSpaces` | `boolean`                          | `false`      | Whether the query may run past the first space |
| `defaultValue` | `TokenFieldValue`                  | empty        | Initial value when uncontrolled                |
| `value`        | `TokenFieldValue`                  | -            | Controlled value                               |
| `onChange`     | `(value: TokenFieldValue) => void` | -            | Called whenever the value changes              |

It returns `value`, `setValue`, `inputRef`, `anchor`, `query`, `trigger`, `isOpen`, `insertToken(text, {appendSpace, value})`, and `close()`. `close()` hides the menu until the next edit or caret move, while Escape keeps it closed for the current trigger. `insertToken` and `close` keep a stable identity, so menu items can capture them in `onAction` without a `dependencies` list.

### PromptInput.Action

Extends HeroUI `Button` props.

| Prop      | Type        | Default | Description              |
| --------- | ----------- | ------- | ------------------------ |
| `tooltip` | `ReactNode` | -       | Optional tooltip content |

### PromptInput.Send

Extends HeroUI `Button` props. Uses the current prompt status to render send, loading, stop, or error icons.

### PromptInput.Shell, Content, Attachments, Toolbar, ToolbarStart, ToolbarEnd, Footer
